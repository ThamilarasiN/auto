function execute-graphQuery($query){
    $batchSize = 100
    $skipResult = 0

    $kqlResult = [System.Collections.Generic.List[object]]::new()
    try{
        while ($true) {
            if ($skipResult -gt 0) {
                $graphResult = Search-AzGraph -Query $query -First $batchSize -SkipToken $graphResult.SkipToken -ErrorAction Stop
            }
            else {
                $graphResult = Search-AzGraph -Query $query -First $batchSize -ErrorAction Stop
            }
                        
            $kqlResult += $graphResult.data
            
            if ($graphResult.data.Count -lt $batchSize) {
                break;
            }
            $skipResult += $skipResult + $batchSize
        }
    }
    catch{
        $msg = $_.Exception.Message
        throw $msg
    }
    return $kqlResult
}
